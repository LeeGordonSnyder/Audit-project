Const PASTE_COL As String = "S"
Const PASTE_FIRST_ROW As Long = 3
Const AUDIT_FIRST_ROW As Long = 7
Const AUDIT_LAST_ROW As Long = 506

Const MARK_FIRST_ROW As Long = 5
Const MARK_LAST_ROW As Long = 254

' Mark In columns: V=UPC, W=Product Name, X=Audit Row, Y=Adjuster Initials, Z=Completed, AA=Date Completed
Const MI_UPC As String = "V": Const MI_NAME As String = "W": Const MI_ROW As String = "X"
Const MI_INIT As String = "Y": Const MI_DONE As String = "Z": Const MI_DATE As String = "AA"
' Mark Out columns: AC=UPC, AD=Product Name, AE=Audit Row, AF=Adjuster Initials, AG=Completed, AH=Date Completed
Const MO_UPC As String = "AC": Const MO_NAME As String = "AD": Const MO_ROW As String = "AE"
Const MO_INIT As String = "AF": Const MO_DONE As String = "AG": Const MO_DATE As String = "AH"

' Overdue-to-Scan columns: AJ=Style/Product Name, AK=Gender, AL=Color, AM=Last Scanned, AN=Days Overdue
Const OD_NAME As String = "AJ": Const OD_GENDER As String = "AK": Const OD_COLOR As String = "AL"
Const OD_LASTSCAN As String = "AM": Const OD_DAYS As String = "AN"
Const OD_FIRST_ROW As Long = 5
Const OD_LAST_ROW As Long = 254
Const OVERDUE_THRESHOLD_DAYS As Long = 29


Private Sub Worksheet_Change(ByVal Target As Range)

    Dim pasteRange As Range, countRange As Range
    Dim miDoneRange As Range, moDoneRange As Range

    Set pasteRange = Me.Range(PASTE_COL & PASTE_FIRST_ROW & ":" & PASTE_COL & Me.Rows.Count)
    Set countRange = Me.Range("H" & AUDIT_FIRST_ROW & ":J" & AUDIT_LAST_ROW)   ' H=Expected Qty, I=BOH, J=Floor
    Set miDoneRange = Me.Range(MI_DONE & MARK_FIRST_ROW & ":" & MI_DONE & MARK_LAST_ROW)
    Set moDoneRange = Me.Range(MO_DONE & MARK_FIRST_ROW & ":" & MO_DONE & MARK_LAST_ROW)

    If Not Intersect(Target, pasteRange) Is Nothing Then
        Call HandlePaste(Target)
    ElseIf Not Intersect(Target, countRange) Is Nothing Then
        Application.EnableEvents = False
        On Error GoTo CleanUpCount
        Dim c As Range
        For Each c In Intersect(Target, countRange).Cells
            Call EvaluateRowForQueue(c.Row)
        Next c
        Application.Calculate
CleanUpCount:
        Application.EnableEvents = True
    ElseIf Not Intersect(Target, miDoneRange) Is Nothing Then
        Application.EnableEvents = False
        Call HandleCheckboxToggle(Intersect(Target, miDoneRange), MI_DATE)
        Application.EnableEvents = True
    ElseIf Not Intersect(Target, moDoneRange) Is Nothing Then
        Application.EnableEvents = False
        Call HandleCheckboxToggle(Intersect(Target, moDoneRange), MO_DATE)
        Application.EnableEvents = True
    End If

End Sub


Private Sub HandlePaste(ByVal Target As Range)

    Dim lastPasted As Long
    lastPasted = Me.Cells(Me.Rows.Count, PASTE_COL).End(xlUp).Row
    If lastPasted < PASTE_FIRST_ROW Then Exit Sub

    Application.EnableEvents = False
    On Error GoTo CleanUp

    Dim curName As String, curSKU As String, curGender As String, curColor As String
    Dim curSize As String, curUPC As String, availVal As String
    Dim addedCount As Long
    Dim nextRow As Long
    nextRow = AUDIT_FIRST_ROW

    Dim r As Long, ln As String
    For r = PASTE_FIRST_ROW To lastPasted
        ln = Trim$(CStr(Me.Cells(r, PASTE_COL).Value))
        If ln <> "" Then
            If LCase$(Left$(ln, 3)) = "sku" Then
                curSKU = Trim$(Mid$(ln, 4))
            ElseIf LCase$(Left$(ln, 4)) = "dept" Then
                curGender = Trim$(Mid$(ln, 5))
            ElseIf LCase$(Left$(ln, 5)) = "style" Then
                ' Style CODE is not stored -- Product Name already serves as the style name.
            ElseIf LCase$(Left$(ln, 5)) = "color" Then
                curColor = Trim$(Mid$(ln, 6))
            ElseIf LCase$(Left$(ln, 4)) = "size" Then
                curSize = Trim$(Mid$(ln, 5))
            ElseIf LCase$(Left$(ln, 3)) = "upc" Then
                curUPC = Trim$(Mid$(ln, 4))
            ElseIf LCase$(Left$(ln, 9)) = "available" Then
                Dim availRaw As String, parts() As String
                availRaw = Trim$(Mid$(ln, 10))
                parts = Split(availRaw, "/")
                If UBound(parts) >= 1 Then
                    availVal = Trim$(parts(1))   ' second number = total on-hand
                Else
                    availVal = Trim$(parts(0))
                End If

                ' "Available" is always the last field for a product -> write the audit line now.
                If curUPC <> "" Then
                    Do While nextRow <= AUDIT_LAST_ROW And Trim$(CStr(Me.Cells(nextRow, "B").Value)) <> ""
                        nextRow = nextRow + 1
                    Loop
                    If nextRow > AUDIT_LAST_ROW Then
                        MsgBox "No blank audit rows left below row " & AUDIT_LAST_ROW & _
                               ". Extend the table's formulas before pasting more.", vbExclamation
                    Else
                        Me.Cells(nextRow, "C").Value = curSKU
                        If curName <> "" Then Me.Cells(nextRow, "D").Value = curName
                        If curGender <> "" Then Me.Cells(nextRow, "E").Value = curGender
                        If curColor <> "" Then Me.Cells(nextRow, "F").Value = curColor
                        If curSize <> "" Then Me.Cells(nextRow, "G").Value = curSize

                        If availVal <> "" And IsNumeric(availVal) Then
                            Me.Cells(nextRow, "H").Value = CDbl(availVal)
                        End If

                        Me.Cells(nextRow, "B").NumberFormat = "@"
                        Me.Cells(nextRow, "B").Value = curUPC   ' written last -- triggers the
                                                                  ' existing Scan Date / Auditor
                                                                  ' Initials formulas already in
                                                                  ' this row, same as a manual scan.
                        addedCount = addedCount + 1

                        ' Negative Expected Qty is a standing data problem, independent of any
                        ' physical count -- flag it for Mark In right away.
                        Call EvaluateRowForQueue(nextRow)

                        nextRow = nextRow + 1
                    End If
                End If

                curName = "": curSKU = "": curGender = "": curColor = "": curSize = "": curUPC = "": availVal = ""
            Else
                ' Unrecognized label -> this is the product name line (start of a new record).
                curName = ln
            End If
        End If
    Next r

    ' Iterative/circular formulas (Scan Date, Auditor Initials) need an explicit
    ' full recalculation to converge -- a single implicit recalc during this loop
    ' isn't reliable for them, unlike ordinary (non-circular) formulas.
    Application.Calculate
    Call RefreshOverdueList

CleanUp:
    Me.Range(PASTE_COL & PASTE_FIRST_ROW & ":" & PASTE_COL & lastPasted).ClearContents
    Application.EnableEvents = True
    If Err.Number <> 0 Then
        MsgBox "Paste parsing hit an error: " & Err.Description, vbExclamation
    ElseIf addedCount > 0 Then
        MsgBox addedCount & " audit line(s) added.", vbInformation
    End If

End Sub


' Checks one audit row after it changes and keeps the Mark In / Mark Out queues in
' sync: queues the row if it now qualifies, and removes it again if it no longer
' does (e.g. a mis-typed count gets corrected) -- UNLESS that entry has already
' been checked off, in which case it's left alone as a completed historical record.
Private Sub EvaluateRowForQueue(ByVal r As Long)

    If r < AUDIT_FIRST_ROW Or r > AUDIT_LAST_ROW Then Exit Sub

    Dim upc As String
    upc = Trim$(CStr(Me.Cells(r, "B").Value))
    If upc = "" Then Exit Sub

    Dim prodName As String
    prodName = CStr(Me.Cells(r, "D").Value)

    Dim qualifiesMarkIn As Boolean, qualifiesMarkOut As Boolean

    Dim expectedQty As Variant
    expectedQty = Me.Cells(r, "H").Value
    If IsNumeric(expectedQty) Then
        If CDbl(expectedQty) < 0 Then qualifiesMarkIn = True
    End If

    Dim status As String
    status = CStr(Me.Cells(r, "M").Value)
    If status = "Overage" Then qualifiesMarkIn = True
    If status = "Shortage" Then qualifiesMarkOut = True

    If qualifiesMarkIn Then
        Call QueueRow(MI_UPC, MI_NAME, MI_ROW, MI_INIT, MI_DONE, r, upc, prodName)
    Else
        Call RemoveFromQueue(MI_UPC, MI_NAME, MI_ROW, MI_INIT, MI_DONE, MI_DATE, r)
    End If

    If qualifiesMarkOut Then
        Call QueueRow(MO_UPC, MO_NAME, MO_ROW, MO_INIT, MO_DONE, r, upc, prodName)
    Else
        Call RemoveFromQueue(MO_UPC, MO_NAME, MO_ROW, MO_INIT, MO_DONE, MO_DATE, r)
    End If

End Sub


Private Sub QueueRow(colUPC As String, colName As String, colRow As String, colInit As String, _
                      colDone As String, ByVal sourceRow As Long, upc As String, prodName As String)

    Dim r As Long
    For r = MARK_FIRST_ROW To MARK_LAST_ROW
        If Trim$(CStr(Me.Cells(r, colRow).Value)) = "" Then Exit For
        If CLng(Me.Cells(r, colRow).Value) = sourceRow Then Exit Sub   ' already queued
    Next r
    If r > MARK_LAST_ROW Then
        MsgBox "The Mark In/Mark Out queue is full (" & MARK_LAST_ROW - MARK_FIRST_ROW + 1 & _
               " rows). Extend it before more discrepancies can be queued.", vbExclamation
        Exit Sub
    End If

    Me.Cells(r, colUPC).NumberFormat = "@"
    Me.Cells(r, colUPC).Value = upc
    Me.Cells(r, colName).Value = prodName
    Me.Cells(r, colRow).Value = sourceRow

    ' Create a real Form Control checkbox over the Completed cell, linked to that
    ' same cell -- clicking it writes TRUE/FALSE to colDone & r, which is what
    ' HandleCheckboxToggle watches for via Worksheet_Change.
    Dim target As Range
    Set target = Me.Range(colDone & r)
    Dim cb As CheckBox
    Set cb = Me.CheckBoxes.Add(target.Left, target.Top, target.Width, target.Height)
    cb.Caption = ""
    cb.LinkedCell = target.Address
    cb.Value = xlOff

End Sub


' Removes a row from a queue (and deletes its checkbox) once it no longer
' qualifies -- but only if it hasn't already been checked off as completed,
' so a corrected mistake never erases an adjuster's completed work record.
Private Sub RemoveFromQueue(colUPC As String, colName As String, colRow As String, colInit As String, _
                             colDone As String, colDate As String, ByVal sourceRow As Long)

    Dim r As Long
    For r = MARK_FIRST_ROW To MARK_LAST_ROW
        Dim v As Variant
        v = Me.Cells(r, colRow).Value
        If Not IsEmpty(v) Then
            If Trim$(CStr(v)) <> "" Then
                If CLng(v) = sourceRow Then
                    If CBool(Me.Cells(r, colDone).Value) = True Then Exit Sub   ' completed -- keep it

                    Dim shp As Shape
                    For Each shp In Me.Shapes
                        On Error Resume Next
                        If shp.TopLeftCell.Address = Me.Cells(r, colDone).Address Then shp.Delete
                        On Error GoTo 0
                    Next shp

                    Me.Cells(r, colUPC).ClearContents
                    Me.Cells(r, colName).ClearContents
                    Me.Cells(r, colRow).ClearContents
                    Me.Cells(r, colInit).ClearContents
                    Me.Cells(r, colDone).ClearContents
                    Me.Cells(r, colDate).ClearContents
                    Exit Sub
                End If
            End If
        End If
    Next r

End Sub


' Fires when a Mark In/Mark Out checkbox's linked cell changes. Stamps today's
' date next to it when checked; clears the date again if unchecked.
Private Sub HandleCheckboxToggle(changedRange As Range, dateCol As String)

    Dim c As Range
    For Each c In changedRange.Cells
        If CBool(c.Value) = True Then
            Me.Cells(c.Row, dateCol).Value = Date
            Me.Cells(c.Row, dateCol).NumberFormat = "mm/dd/yyyy"
        Else
            Me.Cells(c.Row, dateCol).ClearContents
        End If
    Next c

End Sub


' Rebuilds the Overdue-to-Scan list: for every distinct Style + Gender + Color,
' finds its most recent Scan Date across all audit rows, and lists it if that
' date is OVERDUE_THRESHOLD_DAYS or more days old. Run this periodically (it
' also runs automatically after every paste) since staleness is based on
' today's date, not on any single edit.
Public Sub RefreshOverdueList()

    Call EnsureRefreshButton

    Dim dict As Object
    Set dict = CreateObject("Scripting.Dictionary")

    Dim r As Long, key As String
    For r = AUDIT_FIRST_ROW To AUDIT_LAST_ROW
        If Trim$(CStr(Me.Cells(r, "B").Value)) <> "" Then
            If IsDate(Me.Cells(r, "A").Value) Then
                key = CStr(Me.Cells(r, "D").Value) & Chr(9) & CStr(Me.Cells(r, "E").Value) & Chr(9) & CStr(Me.Cells(r, "F").Value)
                Dim scanDate As Date
                scanDate = CDate(Me.Cells(r, "A").Value)
                If dict.Exists(key) Then
                    If scanDate > CDate(dict(key)) Then dict(key) = scanDate
                Else
                    dict.Add key, scanDate
                End If
            End If
        End If
    Next r

    Me.Range(OD_NAME & OD_FIRST_ROW & ":" & OD_DAYS & OD_LAST_ROW).ClearContents

    Dim outRow As Long
    outRow = OD_FIRST_ROW
    Dim k As Variant
    For Each k In dict.Keys
        Dim lastScan As Date
        lastScan = CDate(dict(k))
        Dim daysAgo As Long
        daysAgo = Date - lastScan
        If daysAgo >= OVERDUE_THRESHOLD_DAYS Then
            If outRow <= OD_LAST_ROW Then
                Dim parts() As String
                parts = Split(CStr(k), Chr(9))
                Me.Cells(outRow, OD_NAME).Value = parts(0)
                Me.Cells(outRow, OD_GENDER).Value = parts(1)
                Me.Cells(outRow, OD_COLOR).Value = parts(2)
                Me.Cells(outRow, OD_LASTSCAN).Value = lastScan
                Me.Cells(outRow, OD_LASTSCAN).NumberFormat = "mm/dd/yyyy"
                Me.Cells(outRow, OD_DAYS).Value = daysAgo
                outRow = outRow + 1
            End If
        End If
    Next k

    Application.Calculate

End Sub


Private Sub EnsureRefreshButton()

    Dim b As Button
    For Each b In Me.Buttons
        If b.Name = "btnRefreshOverdue" Then Exit Sub
    Next b

    Dim target As Range
    Set target = Me.Range(OD_LASTSCAN & "3")
    Set b = Me.Buttons.Add(target.Left, target.Top, 150, 18)
    b.Name = "btnRefreshOverdue"
    b.Caption = "Refresh Overdue List"
    b.OnAction = "RefreshOverdueList"

End Sub
